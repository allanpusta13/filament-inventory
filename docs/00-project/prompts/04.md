# Phase 04: Concurrency-Resilient `InventoryService` & `LossLedgerService` (v3.0)

## Objective
Build transactional service classes managing database inventory mutations with pessimistic row locking (`lockForUpdate()`), substitute variant resolution, and omitted cargo shortfall logging.

---

## `app/Services/InventoryService.php`

```php
namespace App\Services;

use App\Models\StockMovement;
use App\Models\TransferRequisition;
use App\Models\TransferRequisitionItem;
use App\Models\LossLedger;
use App\Models\ProductVariant;
use Illuminate\Support\Facades\DB;
use Exception;

class InventoryService
{
    public function recordMovement(
        int $productVariantId,
        int $warehouseId,
        string $type,
        int $baseQuantity,
        ?string $unitName = null,
        int $unitRatio = 1,
        ?int $relatedMovementId = null,
        ?string $referenceType = null,
        ?int $referenceId = null,
        ?string $referenceCode = null
    ): StockMovement {
        if (in_array($type, ['receive', 'transfer_in', 'transit_in', 'adjustment']) && $baseQuantity < 0) {
            throw new Exception("Intake quantity must be positive. Provided: {$baseQuantity}");
        }

        return StockMovement::create([
            'product_variant_id' => $productVariantId,
            'warehouse_id' => $warehouseId,
            'type' => $type,
            'quantity' => $baseQuantity,
            'unit_name_used' => $unitName ?? 'Base Unit',
            'unit_ratio_used' => $unitRatio,
            'related_movement_id' => $relatedMovementId,
            'reference_type' => $referenceType,
            'reference_id' => $referenceId,
            'reference_code' => $referenceCode,
            'created_by' => auth()->id(),
        ]);
    }

    public function executeDirectTransfer(
        int $productVariantId,
        int $fromWarehouseId,
        int $toWarehouseId,
        int $qty,
        string $unitName,
        int $unitRatio,
        ?string $notes = null
    ): void {
        DB::transaction(function () use ($productVariantId, $fromWarehouseId, $toWarehouseId, $qty, $unitRatio, $notes) {
            $baseQty = $qty * $unitRatio;
            $variant = ProductVariant::findOrFail($productVariantId);
            $current = $variant->availableQuantity($fromWarehouseId);

            if ($current < $baseQty) {
                throw new Exception("Insufficient available stock at origin: {$current} base units.");
            }

            $referenceCode = 'DTR-' . date('Ymd') . '-' . strtoupper(uniqid());

            $out = $this->recordMovement(
                $productVariantId, $fromWarehouseId, 'transfer_out', -$baseQty,
                unitName: $unitName, unitRatio: $unitRatio, referenceCode: $referenceCode
            );

            $this->recordMovement(
                $productVariantId, $toWarehouseId, 'transfer_in', $baseQty,
                unitName: $unitName, unitRatio: $unitRatio, relatedMovementId: $out->id, referenceCode: $referenceCode
            );
        });
    }

    public function dispatchRequisition(TransferRequisition $requisition): void
    {
        DB::transaction(function () use ($requisition) {
            foreach ($requisition->items as $item) {
                $targetVariantId = $item->substitute_product_variant_id ?? $item->product_variant_id;
                $dispatchQty = $item->approved_base_qty ?? $item->requested_base_qty;

                $this->recordMovement(
                    $targetVariantId,
                    $requisition->from_warehouse_id,
                    'transit_out',
                    -$dispatchQty,
                    referenceType: TransferRequisition::class,
                    referenceId: $requisition->id,
                    referenceCode: $requisition->reference_code
                );
            }
            $requisition->update(['status' => 'dispatched', 'dispatched_at' => now(), 'dispatched_by' => auth()->id()]);
        });
    }

    public function receiveRequisition(TransferRequisition $requisition, array $receivedQuantities, ?int $receivedBy = null): void
    {
        DB::transaction(function () use ($requisition, $receivedQuantities, $receivedBy) {
            $hasLoss = false;

            foreach ($requisition->items as $item) {
                $targetVariantId = $item->substitute_product_variant_id ?? $item->product_variant_id;
                $shippedQty = $item->shipped_base_qty > 0 ? $item->shipped_base_qty : ($item->approved_base_qty ?? $item->requested_base_qty);
                $received = $receivedQuantities[$item->id] ?? 0;

                if ($received > 0) {
                    $this->recordMovement(
                        $targetVariantId,
                        $requisition->to_warehouse_id,
                        'transit_in',
                        $received,
                        referenceType: TransferRequisition::class,
                        referenceId: $requisition->id,
                        referenceCode: $requisition->reference_code
                    );
                }

                $shortfall = $shippedQty - $received;
                if ($shortfall > 0) {
                    $hasLoss = true;
                    $variant = ProductVariant::findOrFail($targetVariantId);
                    LossLedger::create([
                        'transfer_requisition_id' => $requisition->id,
                        'transfer_requisition_item_id' => $item->id,
                        'product_variant_id' => $targetVariantId,
                        'warehouse_id' => $requisition->to_warehouse_id,
                        'lost_base_qty' => $shortfall,
                        'unit_cost_price' => $variant->cost_price,
                        'total_financial_loss' => $shortfall * $variant->cost_price,
                        'loss_category' => 'shortfall',
                        'recorded_by' => auth()->id(),
                        'recorded_at' => now(),
                    ]);
                }
            }

            $requisition->update([
                'status' => $hasLoss ? 'closed_with_loss' : 'completed',
                'received_by' => $receivedBy ?? auth()->id(),
                'completed_at' => now(),
            ]);
        });
    }
}
```
