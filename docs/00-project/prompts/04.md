### Phase 04: InventoryService Core Engine

#### Objective
Build the transactional service class `App\Services\InventoryService` that orchestrates all database inventory mutations. Enforce strict database transaction isolation, pessimistic locking on row balances (`lockForUpdate()`), and concrete checks to guarantee absolute accounting integrity under heavy concurrency.

---

#### Requirements & Specs

##### 1. Full Core Service Implementation Class (`app/Services/InventoryService.php`)

```php
namespace App\Services;

use App\Models\InTransit;
use App\Models\LossLedger;
use App\Models\ProductVariant;
use App\Models\StockMovement;
use App\Models\TransferRequisition;
use App\Models\WarehouseStock;
use Illuminate\Support\Facades\DB;
use Exception;

class InventoryService
{
    /**
     * Record an atomic stock movement and update the warehouse_stock cache model.
     * Handles concurrent row generation safely.
     */
    public function recordMovement(
        int $variantId,
        int $warehouseId,
        string $type,
        int $baseQuantity,
        ?string $unitName = null,
        int $unitRatio = 1,
        ?string $referenceType = null,
        ?int $referenceId = null,
        ?string $referenceCode = null,
        ?int $relatedMovementId = null
    ): StockMovement {
        // Assert positive parameters for base quantities during intake movements
        if (in_array($type, ['receive', 'transfer_in', 'transit_in', 'adjustment']) && $baseQuantity < 0) {
            throw new Exception("Intake movement quantity must be positive. Provided: {$baseQuantity}");
        }

        return DB::transaction(function () use (
            $variantId, $warehouseId, $type, $baseQuantity,
            $unitName, $unitRatio, $referenceType, $referenceId,
            $referenceCode, $relatedMovementId
        ) {
            // Guardrail 2: Catch Unique Index Race Conditions on Concurrent Creation
            try {
                $stock = WarehouseStock::firstOrCreate(
                    ['variant_id' => $variantId, 'warehouse_id' => $warehouseId],
                    ['on_hand_quantity' => 0, 'reserved_quantity' => 0]
                );
            } catch (\Illuminate\Database\QueryException $e) {
                // Duplicate key error on unique (variant_id, warehouse_id) index handled gracefully
                $stock = WarehouseStock::where('variant_id', $variantId)
                    ->where('warehouse_id', $warehouseId)
                    ->firstOrFail();
            }
            
            // Pessimistic Row Lock
            $stockRow = WarehouseStock::where('id', $stock->id)->lockForUpdate()->first();

            // Verify that decrementing stock does not push physical balance below zero
            if ($baseQuantity < 0 && ($stockRow->on_hand_quantity + $baseQuantity) < 0) {
                throw new Exception("Insufficient stock available for Variant ID {$variantId} at Warehouse ID {$warehouseId}. Available: {$stockRow->on_hand_quantity}, Requested deduction: " . abs($baseQuantity));
            }

            $stockRow->on_hand_quantity += $baseQuantity;
            $stockRow->save();

            return StockMovement::create([
                'variant_id' => $variantId,
                'warehouse_id' => $warehouseId,
                'type' => $type,
                'quantity' => $baseQuantity,
                'unit_name_used' => $unitName ?? 'Base Unit',
                'unit_ratio_used' => $unitRatio,
                'related_movement_id' => $relatedMovementId,
                'reference_type' => $referenceType,
                'reference_id' => $referenceId,
                'reference_code' => $referenceCode,
                'created_by' => auth()->id(),
            ]);
        });
    }

    /**
     * Lock allocated stock when a requisition is mutually confirmed.
     */
    public function lockStockForRequisition(int $requisitionId): void
    {
        DB::transaction(function () use ($requisitionId) {
            $requisition = TransferRequisition::with('items')->findOrFail($requisitionId);

            foreach ($requisition->items as $item) {
                // Guardrail 1: Substitute Variant Resolution
                $actualVariantId = $item->substitute_variant_id ?? $item->variant_id;
                $neededBaseQty = $item->approved_base_qty ?? $item->requested_base_qty;

                // Guardrail 5: Negative Quantities Enforcement
                if ($neededBaseQty <= 0) {
                    throw new Exception("Requisition item {$item->id} contains an invalid base quantity ({$neededBaseQty}).");
                }

                $stock = WarehouseStock::where('variant_id', $actualVariantId)
                    ->where('warehouse_id', $requisition->from_warehouse_id)
                    ->lockForUpdate()
                    ->firstOrFail();

                // Check available stock using formula: on_hand - reserved
                if (($stock->on_hand_quantity - $stock->reserved_quantity) < $neededBaseQty) {
                    throw new Exception("Insufficient unreserved stock for variant ID {$actualVariantId} at origin warehouse.");
                }

                $stock->reserved_quantity += $neededBaseQty;
                $stock->save();
            }

            $requisition->update(['status' => 'confirmed']);
        });
    }

    /**
     * Dispatch Requisition: Release locked stock, decrement origin, and move items into Virtual In-Transit.
     */
    public function dispatchTransfer(int $requisitionId): void
    {
        DB::transaction(function () use ($requisitionId) {
            $requisition = TransferRequisition::with('items')->findOrFail($requisitionId);

            if ($requisition->status !== 'confirmed') {
                throw new Exception("Requisition must be confirmed before dispatch. Current status: {$requisition->status}");
            }

            foreach ($requisition->items as $item) {
                // Guardrail 1: Substitute Variant Resolution
                $actualVariantId = $item->substitute_variant_id ?? $item->variant_id;
                $dispatchQty = $item->approved_base_qty ?? $item->requested_base_qty;

                if ($dispatchQty <= 0) {
                    throw new Exception("Dispatch item {$item->id} contains an invalid base quantity ({$dispatchQty}).");
                }

                $stock = WarehouseStock::where('variant_id', $actualVariantId)
                    ->where('warehouse_id', $requisition->from_warehouse_id)
                    ->lockForUpdate()
                    ->firstOrFail();

                // Adjust balances
                $stock->reserved_quantity = max(0, $stock->reserved_quantity - $dispatchQty);
                $stock->on_hand_quantity = max(0, $stock->on_hand_quantity - $dispatchQty);
                $stock->save();

                // Create dispatch movements
                StockMovement::create([
                    'variant_id' => $actualVariantId,
                    'warehouse_id' => $requisition->from_warehouse_id,
                    'type' => 'transit_out',
                    'quantity' => -$dispatchQty,
                    'unit_name_used' => $item->approved_unit_name ?? $item->requested_unit_name,
                    'unit_ratio_used' => $item->approved_unit_ratio ?? $item->requested_unit_ratio,
                    'reference_type' => TransferRequisition::class,
                    'reference_id' => $requisition->id,
                    'reference_code' => $requisition->reference_code,
                    'created_by' => auth()->id(),
                ]);

                // Register Virtual In-Transit row
                InTransit::create([
                    'requisition_id' => $requisition->id,
                    'requisition_item_id' => $item->id,
                    'variant_id' => $actualVariantId,
                    'dispatched_base_qty' => $dispatchQty,
                    'dispatched_at' => now(),
                    'status' => 'in_transit',
                ]);

                $item->update(['shipped_base_qty' => $dispatchQty]);
            }

            $requisition->update([
                'status' => 'dispatched',
                'dispatched_by' => auth()->id(),
                'dispatched_at' => now(),
            ]);
        });
    }

    /**
     * Scan-to-Receive pipeline with loss accounting and substitute variant matching.
     */
    public function scanToReceive(int $requisitionId, array $receivedItemsData): void
    {
        DB::transaction(function () use ($requisitionId, $receivedItemsData) {
            $requisition = TransferRequisition::with('items')->findOrFail($requisitionId);

            if ($requisition->status !== 'dispatched') {
                throw new Exception("Requisition must be in dispatched state before receiving. Current status: {$requisition->status}");
            }

            $hasLossOrDamage = false;

            foreach ($requisition->items as $item) {
                // Guardrail 1: Substitute Variant Resolution
                $actualVariantId = $item->substitute_variant_id ?? $item->variant_id;
                $expectedBase = $item->shipped_base_qty;

                // Guardrail 6: Prevent Permanent In-Transit Leak (Handle omitted scans)
                if (!isset($receivedItemsData[$item->id])) {
                    // Item was expected but completely omitted from receiving scan (treated as 100% loss)
                    $goodBase = 0;
                    $damagedBase = 0;
                    $lostBase = $expectedBase;
                    $lossCategory = 'Omitted Shipment / Loss in Transit';
                } else {
                    $entry = $receivedItemsData[$item->id];
                    $ratio = $item->approved_unit_ratio ?? $item->requested_unit_ratio;
                    
                    $goodBase = ($entry['good_qty'] ?? 0) * $ratio;
                    $damagedBase = ($entry['damaged_qty'] ?? 0) * $ratio;
                    $lostBase = $expectedBase - ($goodBase + $damagedBase);
                    $lossCategory = $entry['loss_category'] ?? 'Transit Variance';
                }

                // If good items received, debit the target warehouse stock
                if ($goodBase > 0) {
                    $this->recordMovement(
                        variantId: $actualVariantId,
                        warehouseId: $requisition->to_warehouse_id,
                        type: 'transit_in',
                        baseQuantity: $goodBase,
                        unitName: $item->approved_unit_name ?? $item->requested_unit_name,
                        unitRatio: $item->approved_unit_ratio ?? $item->requested_unit_ratio,
                        referenceType: TransferRequisition::class,
                        referenceId: $requisition->id,
                        referenceCode: $requisition->reference_code
                    );
                }

                // Log discrepancy values to LossLedger
                if ($damagedBase > 0 || $lostBase > 0) {
                    $hasLossOrDamage = true;
                    $variant = ProductVariant::findOrFail($actualVariantId);

                    LossLedger::create([
                        'requisition_id' => $requisition->id,
                        'requisition_item_id' => $item->id,
                        'variant_id' => $actualVariantId,
                        'warehouse_id' => $requisition->to_warehouse_id,
                        'lost_base_qty' => max(0, $lostBase),
                        'damaged_base_qty' => $damagedBase,
                        'unit_cost_price' => $variant->cost_price,
                        'total_financial_loss' => (max(0, $lostBase) + $damagedBase) * $variant->cost_price,
                        'loss_category' => $lossCategory,
                        'recorded_by' => auth()->id(),
                        'recorded_at' => now(),
                    ]);
                }

                // Reconcile item base targets
                $item->update([
                    'received_good_base_qty' => $goodBase,
                    'received_damaged_base_qty' => $damagedBase,
                ]);

                // Clear Virtual In-Transit row completely
                InTransit::where('requisition_item_id', $item->id)->update([
                    'status' => 'cleared',
                ]);
            }

            $finalStatus = $hasLossOrDamage ? 'closed_with_loss' : 'completed';

            $requisition->update([
                'status' => $finalStatus,
                'received_by' => auth()->id(),
                'completed_at' => now(),
            ]);
        });
    }
}
```
