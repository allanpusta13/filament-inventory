# Phase 02: Base Inventory Seeder & Demo Workflow Pipeline (v3.0)

## Objective
Build `DemoWorkflowSeeder.php` to seed opening inventory balances directly into the `stock_movements` ledger (using `type => 'receive'`) to initialize physical stock without cached balance tables.

---

## `database/seeders/DemoWorkflowSeeder.php`

```php
namespace Database\Seeders;

use Illuminate\Database\Seeder;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use App\Models\User;
use App\Enums\UserRole;
use Carbon\Carbon;

class DemoWorkflowSeeder extends Seeder
{
    public function run(): void
    {
        $now = Carbon::now();

        // 1. Seed Canonical Users
        $admin = User::create(['name' => 'System Admin', 'email' => 'admin@inventory.com', 'password' => Hash::make('password'), 'role' => UserRole::Admin]);
        $auditor = User::create(['name' => 'Logistics Auditor', 'email' => 'auditor@inventory.com', 'password' => Hash::make('password'), 'role' => UserRole::Auditor]);
        $manilaManager = User::create(['name' => 'Manila Branch Manager', 'email' => 'manila@inventory.com', 'password' => Hash::make('password'), 'role' => UserRole::BranchManager]);
        $cebuManager = User::create(['name' => 'Cebu Branch Manager', 'email' => 'cebu@inventory.com', 'password' => Hash::make('password'), 'role' => UserRole::BranchManager]);

        // 2. Seed Physical Warehouses
        $manilaWhId = DB::table('warehouses')->insertGetId(['code' => 'WH-MNL', 'name' => 'Main Manila Central Hub', 'location' => 'Metropolitan Manila', 'is_active' => true, 'created_at' => $now, 'updated_at' => $now]);
        $cebuWhId = DB::table('warehouses')->insertGetId(['code' => 'WH-CEB', 'name' => 'Cebu Regional Depot', 'location' => 'Mandaue City, Cebu', 'is_active' => true, 'created_at' => $now, 'updated_at' => $now]);

        // 3. User Warehouse Pivot Attachments
        DB::table('user_warehouse')->insert([
            ['user_id' => $manilaManager->id, 'warehouse_id' => $manilaWhId],
            ['user_id' => $cebuManager->id, 'warehouse_id' => $cebuWhId],
        ]);

        // 4. Seed Product Family & Variants
        $productId = DB::table('products')->insertGetId(['name' => 'Specialty Coffee Family', 'category' => 'Beverage Ingredients', 'created_at' => $now, 'updated_at' => $now]);
        
        $variant1Id = DB::table('product_variants')->insertGetId([
            'product_id' => $productId,
            'sku' => 'PROD-COF-500G',
            'barcode' => '4800123456789',
            'name' => '500g Whole Arabica Beans',
            'base_unit_name' => 'gram',
            'cost_price' => 0.0150,
            'sale_price' => 0.0350,
            'reorder_point' => 5000,
            'is_active' => true,
            'created_at' => $now, 'updated_at' => $now
        ]);

        // 5. Seed Unit Conversions
        DB::table('product_variant_unit_conversions')->insert([
            ['product_variant_id' => $variant1Id, 'unit_name' => 'Bag', 'base_unit_ratio' => 1000, 'is_default_purchase' => true, 'is_default_transfer' => false, 'created_at' => $now, 'updated_at' => $now],
            ['product_variant_id' => $variant1Id, 'unit_name' => 'Box', 'base_unit_ratio' => 10000, 'is_default_purchase' => false, 'is_default_transfer' => true, 'created_at' => $now, 'updated_at' => $now],
        ]);

        // 6. Seed Opening Inventory directly into Stock Movements (Pure Derived Stock)
        DB::table('stock_movements')->insert([
            [
                'product_variant_id' => $variant1Id,
                'warehouse_id' => $manilaWhId,
                'type' => 'receive',
                'quantity' => 100000, // 100,000 grams
                'unit_name_used' => 'Box',
                'unit_ratio_used' => 10000,
                'reference_code' => 'INIT-STOCK-MNL',
                'created_by' => $admin->id,
                'created_at' => $now, 'updated_at' => $now
            ],
            [
                'product_variant_id' => $variant1Id,
                'warehouse_id' => $cebuWhId,
                'type' => 'receive',
                'quantity' => 20000, // 20,000 grams
                'unit_name_used' => 'Box',
                'unit_ratio_used' => 10000,
                'reference_code' => 'INIT-STOCK-CEB',
                'created_by' => $admin->id,
                'created_at' => $now, 'updated_at' => $now
            ],
        ]);
    }
}
```
