2. **`RelationManagers/WarehouseStocksRelationManager.php`**:
   * Build this relation table on a custom query against the `ProductVariant` model rather than a direct database relationship.
   * Dynamically fetch and display on-hand, reserved, and available quantities scoped to the active warehouse by calling the derived model methods, highlighting available stock in red if it falls to or below zero.

```php
// app/Filament/Resources/Warehouses/RelationManagers/WarehouseStocksRelationManager.php
namespace App\Filament\Resources\Warehouses\RelationManagers;

use App\Models\ProductVariant;
use Filament\Resources\RelationManagers\RelationManager;
use Filament\Tables\Table;
use Filament\Tables\Columns\TextColumn;

class WarehouseStocksRelationManager extends RelationManager
{
    protected static string $relationship = 'movements'; // Scoped proxy relationship

    public function table(Table $table): Table
    {
        $warehouseId = $this->getOwnerRecord()->id;

        return $table
            ->query(ProductVariant::query()->whereHas('movements', fn ($q) => $q->where('warehouse_id', $warehouseId)))
            ->columns([
                TextColumn::make('sku')->label('PRODUCT SKU')->searchable(),
                TextColumn::make('name')->label('VARIANT NAME'),
                TextColumn::make('on_hand')->label('PHYSICAL ON-HAND')->state(fn ($record) => $record->onHandQuantity($warehouseId))->alignRight(),
                TextColumn::make('reserved')->label('RESERVED (PENDING)')->state(fn ($record) => $record->reservedQuantity($warehouseId))->color('warning')->alignRight(),
                TextColumn::make('available')->label('AVAILABLE')->state(fn ($record) => $record->availableQuantity($warehouseId))->color(fn ($state) => $state <= 0 ? 'danger' : 'success')->bold()->alignRight(),
            ]);
    }
}
```
